# eventfd, described in the syzlang subset defined in tools/syz-differential/README.md.
# Modeled on syzkaller's sys/linux/sys.txt eventfd declarations, restricted to
# the subset grammar (resources, flag sets, typed syscall signatures).

include <linux/eventfd.h>
include <fcntl.h>

resource fd_event[fd]

eventfd_flags = EFD_CLOEXEC, EFD_NONBLOCK, EFD_SEMAPHORE
fcntl_getfd_cmds = F_GETFD

eventfd2(initval int32, flags flags[eventfd_flags]) fd_event
read$eventfd(fd fd_event, buf ptr[out, int64], count len[buf]) int64
write$eventfd(fd fd_event, buf ptr[in, int64], count len[buf]) int64
fcntl$getfd(fd fd_event, cmd flags[fcntl_getfd_cmds]) int32
close$eventfd(fd fd_event)
